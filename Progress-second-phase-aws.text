DONE: Set up scripts for olympics dataset (on small instance)
	cd /wikidata
	cat << EOF > QLever-run
#!/bin/tcsh
cd /wikidata/olympics
/wikidata/qlever/build/qlever-server --index-basename olympics --port 7001 --memory-max-size 142G --cache-max-size 32G --default-query-timeout 600s >& /tmp/QLever   &
/usr/bin/sleep 60s
EOF
	cat << EOF > QLever-stop
#!/bin/tcsh
killall qlever-server
sleep 5s
EOF
	chmod +x QLever-run QLever-stop

DONE: Build Olympics indices (on small instance)
	pip install --break-system-packages qlever
	cd /wikidata/olympics
	echo '{ "ascii-prefixes-only": false, "num-triples-per-batch": 100000 }' > olympics.settings.json
	export FILES="-f /wikidata/olympics/olympics.nt"
	/wikidata/qlever/build/qlever-index --parse-parallel 1 --settings-file olympics.settings.json ${FILES} --index-basename olympics
